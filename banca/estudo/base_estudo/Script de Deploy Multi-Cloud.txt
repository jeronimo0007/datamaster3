
## 3. Script de Deploy Multi-Cloud

**scripts/deploy_azure.sh:**
```bash
#!/bin/bash

echo "🚀 Deployando Solução de Detecção de Fraudes no Azure"
echo "===================================================="

# Variáveis
ENVIRONMENT=${1:-dev}
LOCATION=${2:-brazilsouth}
RESOURCE_GROUP="rg-fraud-detection-${ENVIRONMENT}"

# Colors
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
RED='\033[0;31m'
NC='\033[0m'

# 1. Login no Azure
echo -e "${YELLOW}1. Fazendo login no Azure...${NC}"
az login --use-device-code

# 2. Criar Resource Group
echo -e "${YELLOW}2. Criando Resource Group...${NC}"
az group create \
  --name $RESOURCE_GROUP \
  --location $LOCATION \
  --tags Environment=$ENVIRONMENT Application="FraudDetection"

# 3. Deploy Terraform
echo -e "${YELLOW}3. Deployando infraestrutura com Terraform...${NC}"
cd terraform/azure
terraform init
terraform apply -auto-approve \
  -var="environment=$ENVIRONMENT" \
  -var="location=$LOCATION"

# 4. Configurar Databricks
echo -e "${YELLOW}4. Configurando Azure Databricks...${NC}"
DATABRICKS_WORKSPACE=$(terraform output -raw databricks_workspace_url)
DATABRICKS_TOKEN=$(az account get-access-token --resource 2ff814a6-3304-4ab8-85cb-cd0e6f879c1d --query accessToken -o tsv)

# Upload notebooks
echo -e "${YELLOW}5. Upload de notebooks para Databricks...${NC}"
python3 scripts/upload_databricks_notebooks.py \
  --workspace $DATABRICKS_WORKSPACE \
  --token $DATABRICKS_TOKEN \
  --path notebooks/

# 5. Configurar Data Factory
echo -e "${YELLOW}6. Configurando Azure Data Factory...${NC}"
DATA_FACTORY_NAME=$(terraform output -raw data_factory_name)
az datafactory pipeline create \
  --resource-group $RESOURCE_GROUP \
  --factory-name $DATA_FACTORY_NAME \
  --name "pl_fraud_detection" \
  --pipeline @adf/pipelines/fraud_detection.json

# 6. Configurar Synapse
echo -e "${YELLOW}7. Configurando Azure Synapse...${NC}"
SYNAPSE_NAME=$(terraform output -raw synapse_workspace_name)
sqlcmd -S "$SYNAPSE_NAME.sql.azuresynapse.net" \
  -U sqladmin \
  -P $(terraform output -raw sql_password) \
  -d "fraud_detection" \
  -i sql/synapse_schema.sql

# 7. Deploy da API
echo -e "${YELLOW}8. Deploy da API no Azure App Service...${NC}"
cd ../..
az webapp up \
  --name "fraud-api-$ENVIRONMENT" \
  --resource-group $RESOURCE_GROUP \
  --runtime "PYTHON:3.9" \
  --sku B1

# 8. Configurar Monitoramento
echo -e "${YELLOW}9. Configurando Azure Monitor...${NC}"
az monitor app-insights component create \
  --app "fraud-insights-$ENVIRONMENT" \
  --location $LOCATION \
  --resource-group $RESOURCE_GROUP \
  --application-type web

# 9. Criar Dashboard
echo -e "${YELLOW}10. Criando dashboard no Azure Portal...${NC}"
az portal dashboard create \
  --name "FraudDetectionDashboard" \
  --resource-group $RESOURCE_GROUP \
  --input-path dashboards/azure_dashboard.json

echo -e "\n${GREEN}✅ Deploy concluído com sucesso!${NC}"
echo -e "\n${YELLOW}📊 Recursos criados:${NC}"

echo -e "   Data Lake: ${GREEN}$(terraform output -raw data_lake_name)${NC}"
echo -e "   Databricks: ${GREEN}$DATABRICKS_WORKSPACE${NC}"
echo -e "   Synapse: ${GREEN}$(terraform output -raw synapse_workspace_name)${NC}"
echo -e "   API: ${GREEN}https://fraud-api-$ENVIRONMENT.azurewebsites.net${NC}"
echo -e "   Data Factory: ${GREEN}$DATA_FACTORY_NAME${NC}"

echo -e "\n${YELLOW}🚀 Para iniciar o pipeline:${NC}"
echo -e "   ${GREEN}az datafactory pipeline create-run --resource-group $RESOURCE_GROUP --factory-name $DATA_FACTORY_NAME --name pl_fraud_detection${NC}"
